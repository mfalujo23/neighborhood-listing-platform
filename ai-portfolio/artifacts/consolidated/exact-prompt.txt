export function computeSignature(authToken, url, params) {
  const data = Object.keys(params).sort().reduce((value, key) => value + key + params[key], url);
  return crypto.createHmac('sha1', authToken).update(data, 'utf8').digest('base64');
}

export function verifySignature(providedSig, params, env = process.env, now = Date.now()) {
  if (!providedSig || !env.TWILIO_AUTH_TOKEN || !env.PUBLIC_CALLBACK_URL) return false;
  if (typeof providedSig !== 'string' || providedSig.length > 128) return false;
  const tokens = [env.TWILIO_AUTH_TOKEN];
  const expires = Date.parse(env.TWILIO_TOKEN_OVERLAP_EXPIRES_AT || '');
  const remaining = expires - now;
  if (env.TWILIO_AUTH_TOKEN_NEXT && env.TWILIO_AUTH_TOKEN_NEXT !== env.TWILIO_AUTH_TOKEN
      && Number.isFinite(expires) && remaining > 0 && remaining <= 24 * 60 * 60 * 1000) {
    tokens.push(env.TWILIO_AUTH_TOKEN_NEXT);
  }
  const provided = Buffer.from(providedSig);
  let valid = false;
  for (const token of tokens) {
    const expected = Buffer.from(computeSignature(token, env.PUBLIC_CALLBACK_URL, params));
    const matches = provided.length === expected.length && crypto.timingSafeEqual(provided, expected);
    valid = matches || valid;
  }
  return valid;
