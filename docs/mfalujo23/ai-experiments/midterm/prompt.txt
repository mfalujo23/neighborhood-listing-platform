import 'server-only'

import { createClient } from '@supabase/supabase-js'

export { getSupabaseAdmin } from './supabase-server'

// Session-bound public client. Privileged credentials stay in the server-only
// lazy getter above and are never used for ordinary provider or booking writes.
export function createSupabaseUserClient(authorization?: string) {
  const url = process.env.SUPABASE_URL
  const key = process.env.SUPABASE_ANON_KEY || process.env.SUPABASE_PUBLISHABLE_KEY
  if (!url || !key) throw new Error('Supabase server configuration is missing.')
  return createClient(url, key, {
    ...(authorization ? { global: { headers: { Authorization: authorization } } } : {}),
    auth: { persistSession: false, autoRefreshToken: false },
  })
}

import 'server-only'

import { createClient } from '@supabase/supabase-js'

// Read credentials only when a server request actually needs privileged access.
// Importing an API route during a build must not require production secrets.
export function getSupabaseAdmin() {
  const supabaseUrl =
    process.env.SUPABASE_URL ?? process.env.NEXT_PUBLIC_SUPABASE_URL
  const supabaseServiceKey = process.env.SUPABASE_SERVICE_ROLE_KEY

  if (!supabaseUrl || !supabaseServiceKey) {
    throw new Error('Supabase administrator configuration is missing.')
  }

  return createClient(supabaseUrl, supabaseServiceKey, {
    auth: { persistSession: false, autoRefreshToken: false },
  })
}

import { NextResponse } from 'next/server'
import { ServiceCreateSchema, ServiceListResponseSchema } from '@/schemas/service'
import { createSupabaseUserClient as createUserClient } from '@/lib/supabaseServer'

export async function GET() {
  try {
    const supabase = createUserClient()
    const { data, error } = await supabase
      .from('services')
      .select('id, title, hourly_rate, duration_minutes, coverage_radius_miles')
      .eq('status', 'approved')
      .eq('coverage_active', true)
      .order('created_at', { ascending: false })

    if (error) {
      return NextResponse.json(
        { error: 'Could not retrieve services.' },
        { status: 500 },
      )
    }

    const parsed = ServiceListResponseSchema.safeParse({ data })
    if (!parsed.success) {
      return NextResponse.json({ error: 'Service data is unavailable.' }, { status: 502 })
    }
    return NextResponse.json(parsed.data, { status: 200 })
  } catch {
    return NextResponse.json(
      { error: 'Server configuration or request error.' },
      { status: 500 },
    )
  }
}

export async function POST(request: Request) {
  const authorization = request.headers.get('authorization')
  const bearer = authorization?.match(/^Bearer\s+(.+)$/i)

  if (!bearer || !authorization) {
    return NextResponse.json({ error: 'Sign-in is required.' }, { status: 401 })
  }

  let body: unknown
  try {
    body = await request.json()
  } catch {
    return NextResponse.json(
      { error: 'Request body must be valid JSON.' },
      { status: 400 },
    )
  }

  const parsed = ServiceCreateSchema.safeParse(body)
  if (!parsed.success) {
    return NextResponse.json(
      { error: 'Invalid service data.', details: parsed.error.issues },
      { status: 400 },
    )
  }

  try {
    const supabase = createUserClient(authorization)
    const { data: authData, error: authError } =
      await supabase.auth.getUser(bearer[1])

    if (authError || !authData.user) {
      return NextResponse.json({ error: 'Sign-in is required.' }, { status: 401 })
    }

    const { data: profile, error: profileError } = await supabase
      .from('profiles')
      .select('role')
      .eq('id', authData.user.id)
      .maybeSingle()

    if (
      profileError ||
      !profile ||
      !['provider', 'merchant'].includes(profile.role)
    ) {
      return NextResponse.json(
        { error: 'A provider or merchant account is required.' },
        { status: 403 },
      )
    }

    const { data, error } = await supabase
      .from('services')
      .insert({
        provider_id: authData.user.id,
        ...parsed.data,
        status: 'draft',
      })
      .select(
        'id, provider_id, title, hourly_rate, duration_minutes, coverage_radius_miles, status, created_at',
      )
      .single()

    if (error) {
      const status = error.code === '42501' ? 403 : error.code?.startsWith('23') ? 400 : 500
      return NextResponse.json(
        { error: 'Could not create service.' },
        { status },
      )
    }

    return NextResponse.json({ data }, { status: 201 })
  } catch {
    return NextResponse.json(
      { error: 'Server configuration or request error.' },
      { status: 500 },
    )
  }
}

import { NextResponse } from 'next/server'
import { BookingCreateSchema } from '@/schemas/booking'
import { createSupabaseUserClient } from '@/lib/supabaseServer'

export async function POST(request: Request) {
  let body: unknown

  try {
    body = await request.json()
  } catch {
    return NextResponse.json(
      { error: 'Request body must be valid JSON.' },
      { status: 400 },
    )
  }

  const parsed = BookingCreateSchema.safeParse(body)

  if (!parsed.success) {
    return NextResponse.json(
      { error: 'Invalid booking details.', details: parsed.error.issues },
      { status: 400 },
    )
  }

  const url = process.env.SUPABASE_URL
  const key =
    process.env.SUPABASE_ANON_KEY || process.env.SUPABASE_PUBLISHABLE_KEY

  if (!url || !key) {
    return NextResponse.json(
      { error: 'Server configuration is missing.' },
      { status: 500 },
    )
  }

  try {
    const supabase = createSupabaseUserClient()

    const { error } = await supabase.from('bookings').insert({
      service_id: parsed.data.service_id,
      client_email: parsed.data.client_email,
      requested_date: parsed.data.requested_date,
      notes: parsed.data.notes,
    })

    if (error) {
      console.error('Booking insert failed', { code: error.code })
      const status = error.code === '42501' ? 403 : error.code?.startsWith('23') ? 400 : 503
      return NextResponse.json(
        { error: 'Could not submit booking request.' },
        { status },
      )
    }

    return NextResponse.json(
      { message: 'Booking request submitted.' },
      { status: 201 },
    )
  } catch {
    console.error('Booking route failed')
    return NextResponse.json(
      { error: 'Server configuration or request error.' },
      { status: 500 },
    )
  }
}
